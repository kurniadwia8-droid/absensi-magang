import { neon } from "@neondatabase/serverless"; export default async function handler(req, res) { if (req.method === "OPTIONS") { res.setHeader("Access-Control-Allow-Methods", "GET, POST, OPTIONS"); res.setHeader("Access-Control-Allow-Headers", "Content-Type"); return res.status(200).end(); } if (req.method !== "GET" && req.method !== "POST") { return res.status(405).json({ error: "Metode tidak diizinkan" }); } if (!process.env.DATABASE_URL) { return res.status(500).json({ error: "DATABASE_URL belum diatur di Vercel" }); } try { const sql = neon(process.env.DATABASE_URL); if (req.method === "GET") { const data = await sql` SELECT id, nama, jenis_magang, departemen, tanggal, masuk, pulang FROM attendance_records ORDER BY tanggal DESC, nama ASC `; return res.status(200).json(data); } const { nama, jenis, departemen, tanggal, masuk, pulang } = req.body || {}; if (!nama || !jenis || !departemen || !tanggal) { return res.status(400).json({ error: "Nama, jenis magang, departemen, dan tanggal wajib diisi" }); } const data = await sql` INSERT INTO attendance_records (nama, jenis_magang, departemen, tanggal, masuk, pulang) VALUES (${nama}, ${jenis}, ${departemen}, ${tanggal}, ${masuk || null}, ${pulang || null}) ON CONFLICT (tanggal, nama, jenis_magang, departemen) DO UPDATE SET masuk = COALESCE(attendance_records.masuk, EXCLUDED.masuk), pulang = COALESCE(attendance_records.pulang, EXCLUDED.pulang), updated_at = NOW() RETURNING * `; return res.status(200).json(data[0]); } catch (error) { console.error("API absensi error:", error); return res.status(500).json({ error: "Gagal memproses data absensi" }); } }
